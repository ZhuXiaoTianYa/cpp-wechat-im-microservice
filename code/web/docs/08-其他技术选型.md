## 3. 日期格式化：date-fns

### 使用场景

消息时间的格式化显示。

### 安装

已包含在 dependencies 中：

```json
{
  "date-fns": "^3.6.0"
}
```

### 使用示例

```typescript
import { format, isToday, isYesterday } from 'date-fns';
import { zhCN } from 'date-fns/locale';

export function formatMessageTime(timestamp: number): string {
  const date = new Date(timestamp * 1000);

  if (isToday(date)) {
    return format(date, 'HH:mm');
  }

  if (isYesterday(date)) {
    return `昨天 ${format(date, 'HH:mm')}`;
  }

  return format(date, 'MM-dd HH:mm');
}
```

### 使用示例

### 类型守卫工具

```typescript
// src/utils/proto-helpers.ts
export function isTextMessage(
  content: im_server.MessageContent
): content is im_server.MessageContent & { stringMessage: im_server.StringMessageInfo } {
  return content.messageType === im_server.MessageType.STRING;
}

export function isImageMessage(
  content: im_server.MessageContent
): content is im_server.MessageContent & { imageMessage: im_server.ImageMessageInfo } {
  return content.messageType === im_server.MessageType.IMAGE;
}
```

### bytes 类型处理

```typescript
export function uint8ArrayToBlobURL(bytes: Uint8Array, mimeType: string): string {
  const blob = new Blob([bytes], { type: mimeType });
  return URL.createObjectURL(blob);
}

export function fileToUint8Array(file: File): Promise<Uint8Array> {
  return new Promise((resolve, reject) => {
    const reader = new FileReader();
    reader.onload = () => {
      const arrayBuffer = reader.result as ArrayBuffer;
      resolve(new Uint8Array(arrayBuffer));
    };
    reader.onerror = reject;
    reader.readAsArrayBuffer(file);
  });
}
```
