## 核心文件说明

### 1. 入口文件

**src/main.tsx**

```typescript
import React from 'react';
import ReactDOM from 'react-dom/client';
import App from './App';
import './styles/globals.css';

ReactDOM.createRoot(document.getElementById('root')!).render(
  <React.StrictMode>
    <App />
  </React.StrictMode>
);
```

### 2. 根组件

**src/App.tsx**

```typescript
import { BrowserRouter } from 'react-router';
import { Toaster } from 'sonner';
import AppRoutes from './routes';

export default function App() {
  return (
    <BrowserRouter>
      <AppRoutes />
      <Toaster position="top-center" />
    </BrowserRouter>
  );
}
```

### 3. 路由配置

**src/routes.tsx**

```typescript
import { Routes, Route, Navigate } from 'react-router';
import { useIMStore } from '@/store/useIMStore';
import LoginPage from '@/pages/LoginPage';
import RegisterPage from '@/pages/RegisterPage';
import MainLayout from '@/pages/MainLayout';
import ChatPage from '@/pages/ChatPage';
import ContactsPage from '@/pages/ContactsPage';
import ProfilePage from '@/pages/ProfilePage';

export default function AppRoutes() {
  const sessionId = useIMStore((state) => state.sessionId);

  return (
    <Routes>
      <Route path="/login" element={<LoginPage />} />
      <Route path="/register" element={<RegisterPage />} />
      <Route path="/" element={<Navigate to="/login" />} />

      <Route
        path="/app"
        element={sessionId ? <MainLayout /> : <Navigate to="/login" />}
      >
        <Route index element={<Navigate to="/app/chat" />} />
        <Route path="chat" element={<ChatPage />} />
        <Route path="chat/:id" element={<ChatPage />} />
        <Route path="contacts" element={<ContactsPage />} />
        <Route path="profile" element={<ProfilePage />} />
      </Route>

      <Route path="*" element={<Navigate to="/" />} />
    </Routes>
  );
}
```

## 命名规范

### 文件命名

- **组件文件**: PascalCase，如 `MessageBubble.tsx`
- **工具文件**: kebab-case，如 `proto-helpers.ts`
- **Hook 文件**: camelCase，如 `useWebSocket.ts`

### 代码命名

- **组件**: PascalCase，如 `MessageBubble`
- **函数**: camelCase，如 `formatMessageTime`
- **常量**: UPPER_SNAKE_CASE，如 `MAX_MESSAGE_LENGTH`
- **接口**: PascalCase，如 `IMessageProps`

## 代码组织原则

### 1. 单一职责

每个文件/组件只负责一个功能：

```typescript
// 好的做法
// MessageBubble.tsx - 只负责渲染消息气泡
// ChatArea.tsx - 只负责聊天区域
// SessionList.tsx - 只负责会话列表
```

### 2. 依赖注入

通过 props 传递依赖：

```typescript
// 好的做法
function MessageBubble({ message, onDelete }: Props) {
  return <div onClick={() => onDelete(message.messageId)}>...</div>;
}
```

### 3. 状态管理

使用 Zustand 集中管理状态：

```typescript
// 在 Store 中定义
addMessage: (sessionId, msg) => set((state) => {
  state.messages.get(sessionId)?.push(msg);
});

// 在组件中使用
const addMessage = useIMStore((state) => state.addMessage);
```
