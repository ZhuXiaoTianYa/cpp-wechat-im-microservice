64 virtual void
65 GetTransmitTarget(::google::protobuf::RpcController *controller,
66 const ::im_server::NewMessageReq *request,
67 ::im_server::GetTransmitTargetRsp *response,
68 ::google::protobuf::Closure *done) {
69 brpc::ClosureGuard rpc_guard(done);
70 auto err_response = [this,
71 response](const std::string &rid,
72 const std::string &errmsg) -> void {
73 response->set_request_id(rid);
74 response->set_success(false);
75 response->set_errmsg(errmsg);
76 return;
77 };
78 std::string rid = request->request_id();
79 std::string uid = request->user_id();
80 std::string chat_ssid = request->chat_session_id();
81 const MessageContent &content = request->message();
82
83 LOG_INFO("RPC GetTransmitTarget | request_id={} | user_id={} | session_id={} | msg_type={}",
84 rid, uid, chat_ssid, content.message_type());
85
86 auto channel = _mm_channels->choose(_user_service_name);
87 if (!channel) {
88 LOG_ERROR("RPC GetTransmitTarget | request_id={} | 阶段=RPC选路 | 结果=失败 | target={}",
89 rid, _user_service_name);
90 return err_response(rid, "未找到用户管理子服务节点");
91 }
92
93 UserService_Stub stub(channel.get());
94 GetUserInfoReq req;
95 GetUserInfoRsp resp;
96 brpc::Controller cntl;
97 req.set_request_id(rid);
98 req.set_user_id(uid);
99 stub.GetUserInfo(&cntl, &req, &resp, nullptr);
100 if (cntl.Failed() || resp.success() == false) {
101 LOG_ERROR("RPC GetTransmitTarget | request_id={} | 阶段=拉取发送者信息 | 结果=失败 | brpc={}",
102 rid, cntl.ErrorText());
103 return err_response(rid, "用户子服务调用失败");
104 }
105
106 MessageInfo message;
107 message.set_message_id(uuid());
108 message.set_chat_session_id(chat_ssid);
109 message.set_timestamp(time(nullptr));
110 message.mutable_sender()->CopyFrom(resp.user_info());
111 message.mutable_message()->CopyFrom(request->message());
112
113 std::string data = message.SerializeAsString();
114 LOG_DEBUG("RPC GetTransmitTarget | request_id={} | 阶段=序列化消息 | size={}", rid, data.size());
115
116 bool ret = _mq_client->publish(_exchange_name, data, _routing_key);
117 if (ret == false) {
118 LOG_ERROR("RPC GetTransmitTarget | request_id={} | 阶段=MQ发布 | 结果=失败 | exchange={} | routing_key={}",
119 rid, _exchange_name, _routing_key);
120 return err_response(rid, "持久化消息发布失败");
121 }
122
123 auto target_list = _mysql_chat_session_member_table->members(chat_ssid);
124 LOG_INFO("RPC GetTransmitTarget | request_id={} | 阶段=完成 | target_count={}", rid, target_list.size());
125
126 for (auto &id : target_list) {
127 response->add_target_id_list(id);
128 }
129 response->set_request_id(rid);
130 response->set_success(true);
131 response->mutable_message()->CopyFrom(message);
132 }

134private:
135 ChatSessionMemberTable::ptr _mysql_chat_session_member_table;
136 std::string _exchange_name;
137 std::string _routing_key;
138 MQClient::ptr _mq_client;
139 std::string _user_service_name;
140 ServiceManager::ptr _mm_channels;

267 void make_rpc_object(const uint16_t &port, const uint32_t &timeout,
268 const uint8_t &num_threads) {
269 if (!_mysql_client) {
270 LOG_ERROR("转发服务启动检查失败 | 组件=MySQL | 原因=未初始化");
271 abort();
272 }
273 if (!_mm_channels) {
274 LOG_ERROR("转发服务启动检查失败 | 组件=下游RPC信道 | 原因=未初始化");
275 abort();
276 }
277 if (!_mq_client) {
278 LOG_ERROR("转发服务启动检查失败 | 组件=RabbitMQ | 原因=未初始化");
279 abort();
280 }
281 if (!_service_discoverer) {
282 LOG_ERROR("转发服务启动检查失败 | 组件=etcd服务发现 | 原因=未初始化");
283 abort();
284 }
285 _rpc_server = std::make_shared<brpc::Server>();
286 TransmiteServiceImpl *transmite_service = new TransmiteServiceImpl(
287 _mysql_client, _exchange_name, _routing_key, _mq_client,
288 _mm_channels, _user_service_name);
289 int ret = _rpc_server->AddService(
290 transmite_service, brpc::ServiceOwnership::SERVER_OWNS_SERVICE);
291 if (ret == -1) {
292 LOG_ERROR("转发服务启动失败 | 阶段=添加RPC服务");
293 abort();
294 }
295 brpc::ServerOptions options;
296 options.idle_timeout_sec = timeout;
297 options.num_threads = num_threads;
298 ret = _rpc_server->Start(port, &options);
299 if (ret == -1) {
300 LOG_ERROR("转发服务启动失败 | 阶段=启动RPC服务器 | port={}", port);
301 abort();
302 }
303 LOG_INFO("转发服务启动成功 | port={} | timeout={}s | threads={}", port, timeout, num_threads);
304 }
