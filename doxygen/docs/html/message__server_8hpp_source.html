375 void onMessage(const char *body, size_t sz) {
376 LOG_DEBUG("MQ消费 | body_size={}", sz);
377
378 MessageInfo message;
379 bool ret = message.ParseFromArray(body, sz);
380 if (ret == false) {
381 LOG_ERROR("MQ消费 | 阶段=反序列化消息 | 结果=失败");
382 return;
383 }
384
385 std::string msg_id = message.message_id();
386 std::string session_id = message.chat_session_id();
387 std::string sender_id = message.sender().user_id();
388 MessageType msg_type = message.message().message_type();
389
390 LOG_INFO("MQ消费 | msg_id={} | session_id={} | sender={} | type={}",
391 msg_id, session_id, sender_id, msg_type);
392
393 std::string file_id, file_name, content;
394 int64_t file_size;
395 switch (message.message().message_type()) {
396 case MessageType::STRING:
397 content = message.message().string_message().content();
398 ret = _es_message->appendData(
399 message.sender().user_id(), message.message_id(),
400 message.timestamp(), message.chat_session_id(), content);
401 if (ret == false) {
402 LOG_ERROR("MQ消费 | msg_id={} | 阶段=ES写入文本消息 | 结果=失败", msg_id);
403 return;
404 }
405 LOG_DEBUG("MQ消费 | msg_id={} | 阶段=ES写入文本消息 | 结果=成功", msg_id);
406 break;
407 case MessageType::IMAGE: {
408 const auto &msg = message.message().image_message();
409 file_size = msg.image_content().size();
410 LOG_DEBUG("MQ消费 | msg_id={} | 阶段=处理图片消息 | size={}", msg_id, file_size);
411 ret = _PutFile("", msg.image_content(), file_size, file_id);
412 if (ret == false) {
413 LOG_ERROR("MQ消费 | msg_id={} | 阶段=上传图片到文件服务 | 结果=失败", msg_id);
414 return;
415 }
416 } break;
417 case MessageType::FILE: {
418 const auto &msg = message.message().file_message();
419 file_name = msg.file_name();
420 file_size = msg.file_size();
421 LOG_DEBUG("MQ消费 | msg_id={} | 阶段=处理文件消息 | name={} | size={}", msg_id, file_name, file_size);
422 ret = _PutFile(file_name, msg.file_contents(), file_size, file_id);
423 if (ret == false) {
424 LOG_ERROR("MQ消费 | msg_id={} | 阶段=上传文件到文件服务 | 结果=失败", msg_id);
425 return;
426 }
427 } break;
428 case MessageType::SPEECH: {
429 const auto &msg = message.message().speech_message();
430 file_size = msg.file_contents().size();
431 LOG_DEBUG("MQ消费 | msg_id={} | 阶段=处理语音消息 | size={}", msg_id, file_size);
432 ret = _PutFile("", msg.file_contents(), file_size, file_id);
433 if (ret == false) {
434 LOG_ERROR("MQ消费 | msg_id={} | 阶段=上传语音到文件服务 | 结果=失败", msg_id);
435 return;
436 }
437 } break;
438 default:
439 LOG_ERROR("MQ消费 | msg_id={} | 消息类型未知 | type={}", msg_id, msg_type);
440 return;
441 }
442 Message msg(message.message_id(), message.chat_session_id(),
443 message.sender().user_id(),
444 message.message().message_type(),
445 boost::posix_time::from_time_t(message.timestamp()));
446 msg.file_id(file_id);
447 msg.file_name(file_name);
448 msg.file_size(file_size);
449 msg.content(content);
450 ret = _mysql_message->insert(msg);
451 if (ret == false) {
452 LOG_ERROR("MQ消费 | msg_id={} | 阶段=MySQL写入消息 | 结果=失败", msg_id);
453 return;
454 }
455
456 LOG_INFO("MQ消费 | msg_id={} | 阶段=完成 | file_id={}", msg_id, file_id);
457 }

485 stub.GetMultiFile(&cntl, &req, &rsp, nullptr);

487 LOG_ERROR("文件子服务调用失败 - {}", cntl.ErrorText());

489 }

491 for (auto it = fmap.begin(); it != fmap.end(); it++) {
492 file_data_list.insert(

671 void make_rpc_object(const uint16_t &port, const uint32_t &timeout,

675 abort();
676 }

678 LOG_ERROR("消息服务启动检查失败 | 组件=Elasticsearch | 原因=未初始化");
679 abort();
680 }
681 if (!_mm_channels) {
682 LOG_ERROR("消息服务启动检查失败 | 组件=下游RPC信道 | 原因=未初始化");
683 abort();
684 }
685 if (!_service_discoverer) {
686 LOG_ERROR("消息服务启动检查失败 | 组件=etcd服务发现 | 原因=未初始化");
687 abort();
688 }
689 _rpc_server = std::make_shared<brpc::Server>();
690
691 MessageServiceImpl *message_service =
692 new MessageServiceImpl(_es_client, _mysql_client, _mm_channels,
693 _file_service_name, _user_service_name);
694 int ret = _rpc_server->AddService(
695 message_service, brpc::ServiceOwnership::SERVER_OWNS_SERVICE);
696 if (ret == -1) {
697 LOG_ERROR("消息服务启动失败 | 阶段=添加RPC服务");
698 abort();
699 }
700 brpc::ServerOptions options;
701 options.idle_timeout_sec = timeout;
702 options.num_threads = num_threads;
703 ret = _rpc_server->Start(port, &options);
704 if (ret == -1) {
705 LOG_ERROR("消息服务启动失败 | 阶段=启动RPC服务器 | port={}", port);
706 abort();
707 }
708 LOG_INFO("消息服务启动成功 | port={} | timeout={}s | threads={}", port, timeout, num_threads);
709
710 auto callback =
711 std::bind(&MessageServiceImpl::onMessage, message_service,
712 std::placeholders::_1, std::placeholders::_2);
713 _mq_client->consume(_queue_name, callback);
714 LOG_INFO("消息服务MQ消费启动 | queue={}", _queue_name);
715 }
