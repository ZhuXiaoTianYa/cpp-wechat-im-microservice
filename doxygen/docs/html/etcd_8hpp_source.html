104 int sz = resp.keys().size();

111 }

119 void callback(const etcd::Response &resp)
120 {
121 if (resp.is_ok() == false)
122 {
123 LOG_ERROR("收到一个错误的事件通知: {}", resp.error_message());
124 return;
125 }
126 for (auto &ev : resp.events())
127 {
128 if (ev.event_type() == etcd::Event::EventType::PUT)
129 {
130 if (_put_cb)
131 _put_cb(ev.kv().key(), ev.kv().as_string());
132 LOG_DEBUG("新增服务: {}-{}", ev.kv().key(), ev.kv().as_string());
133 }
134 else if (ev.event_type() == etcd::Event::EventType::DELETE_)
135 {
136 if (_del_cb)
137 _del_cb(ev.prev_kv().key(), ev.prev_kv().as_string());
138 LOG_DEBUG("下线服务: {}-{}", ev.prev_kv().key(), ev.prev_kv().as_string());
139 }
140 }
141 }
