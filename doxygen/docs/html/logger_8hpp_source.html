46inline void init_logger(bool mode, std::string &filename, uint32_t level) {
47 if (mode == false) {
48 g_default_logger = spdlog::stdout_color_mt("default-logger");
49 g_default_logger->set_level(spdlog::level::trace);
50 g_default_logger->flush_on(spdlog::level::trace);

52 g_default_logger =

54 filename);

56 g_default_logger->flush_on((spdlog::level::level_enum)level);
