100 bool client(const server_t::connection_ptr &conn, std::string &uid,

112 return true;

120 void remove(const server_t::connection_ptr &conn) {
121 std::unique_lock<std::mutex> lock(_mutex);
122 auto it = _conn_clients.find(conn);
123 if (it == _conn_clients.end()) {
124 LOG_WARN("按连接句柄解析客户端信息失败 | conn={}", (void *)conn.get());
125 return;
126 }
127 const std::string removed_uid = it->second.uid;
128 _uid_connections.erase(removed_uid);
129 _conn_clients.erase(it);
130 LOG_DEBUG("已移除 WebSocket 连接映射 | user_id={}", removed_uid);
131 return;
132 }
