38 MQClient(const std::string &user, const std::string &passwd,

69 void declareComponents(

73 _channel->declareExchange(exchange, AMQP::ExchangeType::direct)

97 bool publish(const std::string &exchange, const std::string &msg,
98 const std::string &routing_key = "routing_key") {
99 LOG_DEBUG("向交换机{}-{}发送消息", exchange, routing_key);
100 bool ret = _channel->publish(exchange, routing_key, msg);
101 if (ret == false) {

108 void consume(const std::string &queue, const MessageCallback &cb) {

111 _channel->consume(queue, "consume-tag")
112 .onReceived([this, cb](const AMQP::Message &message,
113 uint64_t deliveryTag, bool redelivered) {
114 cb(message.body(), message.bodySize());
115 _channel->ack(deliveryTag);

117 .onError([queue](const char *message) {
118 LOG_ERROR("订阅{}消息失败: {}", queue, message);
119 exit(0);
120 });
121 }
